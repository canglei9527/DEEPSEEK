# Idle Battery Saver 配置文件
# 修改后重启手机（或停用再启用模块）生效

# ---------- 息屏限频 ----------
# 息屏时"超大核"最高频率上限 (kHz)，0 = 不限频
# 骁龙8至尊版超大核默认最高 4320000 (4.32GHz)
# 建议区间 1500000 ~ 2200000；若息屏听歌/下载明显变卡，调大或直接设 0
BIG_MAX_OFF=2016000

# 息屏 GPU 最高频率上限 (kHz)，0 = 关闭（默认关，收益较小）
# 仅支持 kgsl 节点机型；节点不存在会自动跳过
GPU_MAX_OFF=0

# ---------- Doze 加速 / 扫描 ----------
# Doze 加速：1=开 0=关
# DOZE_INACTIVE_SEC：息屏后进入 Doze 状态机的等待时长（秒），
# 精确到秒，范围 30 ~ 3600，默认 300（5分钟）。例：120 = 2分钟
# DOZE_PER_APP：1=Doze 只冻结受限应用（自动豁免所有非受限第三方应用，
# 微信等完全不受影响）；0=全局 Doze（所有应用都受影响）
DOZE_ACCEL=1
DOZE_INACTIVE_SEC=300
DOZE_PER_APP=1

# Doze 深度档：0=轻度（原生系统策略） 1=中度激进 2=极致激进
# 与「进入 Doze 等待时长」相互独立（一个管多久睡，一个管睡多死）：
# v3.37 起深度档只收紧受限应用的冻结（更快进 deep idle、维护窗口更短、
# light idle 更长）；alarm 参数固定原生 10 分钟，白名单应用（微信/QQ 等）
# 的推送唤醒频率不受深度档影响，消息照常及时。
DOZE_DEEP=0

# 关闭 WiFi / 蓝牙「始终扫描」：1=开 0=关
SCAN_OFF=1

# ---------- 息屏传感器停止（可选） ----------
# 1=息屏期间停止"不重要的"手势类传感器监听（拿起唤醒/双击亮屏/AOD 手势），
# 亮屏自动恢复；保留接近传感器（通话自动息屏）与光线传感器（自动亮度）。
# 只处理存在且原本开启的项，不影响常显 AOD / 来电 / 通知声音。
SENSOR_OFF=0

# 息屏内核追踪停止（v3.38）：1=息屏进 Doze 后关闭 tracing_on（ftrace 记录），
# 亮屏自动恢复原值；原本已关闭（0）的设备不动（避免亮屏后误开）；0=关闭此功能
TRACE_OFF=1

# ---------- 免生效时间段（重点新功能） ----------
# 在该时间段内，模块的【所有功能】暂停：Doze 参数、扫描、限频、App Standby 全部还原，
# 离开时间段后自动恢复。格式 HH:MM，支持跨天。
# 示例：晚上 22:00 ~ 早上 07:00 不生效 → INACTIVE_START=22:00 / INACTIVE_END=07:00
# 留空 = 不启用
INACTIVE_START=
INACTIVE_END=

# ---------- App Standby（可选） ----------
# 模块生效期间，把下列应用设为 restricted 桶（系统级抑制后台唤醒/任务）
# 只放"不常用、无推送需求"的应用，微信/QQ/钉钉等别放！
# 空格分隔多个包名。卸载/停用/窗口期内自动还原原桶位。
RESTRICT_PKGS=""

# ---------- KSU 原生 WebUI ----------
# 在 KernelSU Manager → 模块 → 本模块详情 → WebUI 中打开实时状态页
# 页面通过 KSU 注入的 ksu.exec bridge 读取状态，无需额外服务/端口
# v3.36：1=WebUI 打开时每 5 秒按需生成状态（页面关闭不写盘）；0=关闭状态生成
WEBUI_EN=1

# ---------- 其他 ----------
# v3.35：屏幕状态检测固定 5 秒一次；息屏限制（传感器/限频/关核）与 Doze 共用
# 同一套倒计时（DOZE_INACTIVE_SEC），息屏后倒计时结束与进入 Doze 同步生效；
# POLL 仅控制 Doze 参数守护的检查节拍（秒），建议 10 ~ 30
POLL=15

# ---------- 系统清理（v3.42 合并自「系统清理」模块） ----------
# 总开关：1=激活 0=未激活。刷入时可按音量键选择（音量+激活/音量-不激活，
# 10 秒无操作默认不激活）；也可在 WebUI「系统清理」区块随时手动开启/关闭。
# 激活后提供：一键清理（ANR/墓碑/系统日志/HyperOS 日志/系统缓存/崩溃记录/
# 云同步缓存/内核调试日志/APP缓存修剪）+ 每日定时 + 开机自动清理 +
# 诊断日志开关。所有清理只删缓存/日志类文件，不动应用数据。
CLEANER_EN=1
# 每日自动清理：1=开启（每天 CLEAN_TIME 时刻执行） 0=关闭
AUTO_CLEAN=0
# 每日自动清理时刻（HH:MM，24 小时制，默认 04:00 凌晨）
CLEAN_TIME=04:00
# 开机自动清理：1=开机后约 5 分钟自动清理一次 0=关闭
BOOT_CLEAN=0
# 系统诊断日志：1=保持开启（系统默认，可反馈 bug） 0=关闭（减少日志堆积，
# 更省电省空间；注意关闭后系统 bug 日志变少，排查问题不便）
DIAG_LOG=1
DISABLE_CORES="6 7"
